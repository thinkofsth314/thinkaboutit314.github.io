#include <iostream>
#include <string>
using namespace std;

template <class T>
struct DNode{
    T data;
    DNode<T>* next;
};

template <class T>
class LinkStack{
    private:
        LinkList<T> L;
    public:
        LinkStack(){

        }
        ~LinkStack(){

        }
        int getLength(){
            return L.getLength();
        }
        bool isEmpty(){
            return L.isEmpty();
        }
        void push(T x){
            L.insertEleAtPos(0,x);
        }

            return L.deleteEleAtPos(0);
        }

            return L.getEleAtPos(0);
        }

template <class T>
class LinkQueue{
    private:
        DNode<T>* front;
        DNode<T>* back;
        int length;
    public:
        LinkQueue(){
            front = new DNode<T>;
            front->next = nullptr;
            back = front;
            length = 0;
        }
        ~LinkQueue(){
            while(front!=NULL){
                back = front;
                front = front->next;
                delete back;
            }
        }
        int getLength(){
            return length;
        }
        bool isEmpty(){
            return length==0; // return front==back;
        }
        void enQueue(T x){
            DNode<T>* newNode = new DNode<T>;
            newNode->data = x;
            newNode->next = nullptr;
            back->next = newNode;
            back = newNode;
            length++;
        }
        T deQueue(){
            if(length==0) throw "Queue Empty";
            DNode<T>* delNode = front->next;
            T delData = delNode->data;
            front->next = delNode->next;
            length--;
            delete delNode;
            if(length==0) back = front;
            return delData;
        }
        T peekQueue(){
            if(length==0) throw "Queue Empty";
            return front->next->data;
        }
        void printQueue(){
            DNode<T>* tmpN = front->next;
            cout << "Queue front: ";
            while(tmpN!=NULL){
                cout << tmpN->data << " ";
                tmpN = tmpN->next;
            }
            cout << ": queue back" << endl;
        }
};
