## 链表(实现队列和栈）
``` cpp
#include <iostream>
#include <string>
using namespace std;

template <class T>
struct DNode{
    T data;
    DNode<T>* next;
};

template <class T>
class LinkList{
    private:
        int length;
        DNode<T>* head;
    public:
        LinkList(){
            length = 0;
            head = new DNode<T>;
            head->next = nullptr; // head->next = NULL;
        }
        LinkList(T* a, int n){
            length = 0;
            head = new DNode<T>;
            head->next = nullptr;
            for(int i=n-1;i>=0;i--){
                insertEleAtPos(0,a[i]);
            }
        }
        ~LinkList(){
            DNode<T>* tmpP = head;
            while(head!=NULL){
                tmpP = head;
                head = tmpP->next;
                delete tmpP;
            }
        }
        int getLength(){
            return length;
        }
        bool isEmpty(){
            return length==0; // return head->next==NULL;
        }
        void insertEleAtPos(int i, T x){
            if(i<0 || i > length){
                throw "Illegal Position";
            }
            DNode<T>* tmpP = head;
            int j = -1;
            while(j!=i-1){
                tmpP = tmpP->next;
                j++;
            }
            DNode<T>* newP = new DNode<T>;
            newP->data = x;
            newP->next = tmpP->next; // point to a_i
            tmpP->next = newP;
            length++;
        }
        T deleteEleAtPos(int i){
            if(i<0 || i >= length){
                throw "Illegal Position";
            }
            DNode<T>* tmpP = head;
            int j = -1;
            while(j!=i-1){
                tmpP = tmpP->next;
                j++;
            }
            DNode<T>* delP = tmpP->next;
            tmpP->next = delP->next;
            T delData = delP->data;
            delete delP;
            length--;
            return delData;
        }
        void setEleAtPos(int i, T x){
            if(i<0 || i >= length){
                throw "Illegal Position";
            }
            DNode<T>* tmpP = head;
            int j = -1;
            while(j!=i){
                tmpP = tmpP->next;
                j++;
            }
            tmpP->data = x;
        }
        T getEleAtPos(int i){
            if(i<0 || i >= length){
                throw "Illegal Position";
            }
            DNode<T>* tmpP = head;
            int j = -1;
            while(j!=i){
                tmpP = tmpP->next;
                j++;
            }
            return tmpP->data;
        }
        int locateEle(T x){
            DNode<T>* tmpP = head->next;
            int j = 0;
            while(tmpP!=NULL){
                if(tmpP->data==x){
                    return j;
                }
                tmpP = tmpP->next;
                j++;
            }
            return -1;
        }
        void printList(){
            DNode<T>* tmpP = head->next;
            while(tmpP!=NULL){
                cout << tmpP->data << " ";
                tmpP = tmpP->next;
            }
        }
};
```

template <class T>

        }

            return L.getLength();
        }
        bool isEmpty(){
            return L.isEmpty();
        }
        void push(T x){
            L.insertEleAtPos(0,x);
        }
        T pop(){
            return L.deleteEleAtPos(0);

            return L.getEleAtPos(0);

## 队列
``` cpp
template <class T>
class LinkQueue{
    private:
        DNode<T>* front;
        DNode<T>* back;
        int length;
    public:
        LinkQueue(){
            front = new DNode<T>;
            front->next = nullptr;
            back = front;
            length = 0;
        }
        ~LinkQueue(){
            while(front!=NULL){
                back = front;
                front = front->next;
                delete back;
            }
        }
        int getLength(){
            return length;
        }
        bool isEmpty(){
            return length==0; // return front==back;
        }
        void enQueue(T x){
            DNode<T>* tmpN = new DNode<T>;
            tmpN->data = x;
            tmpN->next = nullptr;
            back->next = tmpN;
            back = tmpN;
            length++;
        }
        T deQueue(){
            if(length==0) throw "Queue Empty!";
            DNode<T>* tmpP = front->next;
            front->next= tmpP->next;
            T tmpD = tmpP->data;
            delete tmpP;
            length--;
            if(length==0) back = front;
            return tmpD;
        }
        T peekQueue(){
            if(length==0) throw "Queue Empty!";
            return front->next->data;
        }
        void printQueue(){
            cout << "Queue front: ";
            DNode<T>* tmpP = front->next;
            while(tmpP != NULL){
                cout << tmpP->data << " ";
                tmpP = tmpP->next;
            }
            cout << ": queue end" << endl;
        }
};
```
