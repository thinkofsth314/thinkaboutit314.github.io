``` cpp
#include<iostream>
using namespace std;
#define MAXSIZE 100
template <typename T>//T的类型未知，在main函数里给出
class Veclist
{
    private:
        int capacity;
        int length;
        T* arr;
        void doubleListSize(){
            int newCapacity=capacity*2;
            T* newArr=new T[newCapacity];
            for(int i=0;i<length;i++){
                newArr[i]=arr[i];
            }
            delete[] arr;
            arr=newArr;
            capacity=newCapacity;
        }
    public:
        Veclist(){//构造函数
            length=0;
            capacity=MAXSIZE;
            arr=new T[MAXSIZE];
        }
        //用外部传入的数据初始化列表
        Veclist(T* a,int n){
            length=n;
            //如果传入的元素个数大于MAXMIZE，则按照2倍分配
            capacity=(n>MAXSIZE)?2*n:MAXSIZE;
            arr=new T[capacity];
            for(int i=0;i<n;i++){
                arr[i]=a[i];
            }
        }
        ~Veclist(){//析构函数
            delete[] arr;
        }
        int getLength(){
            return length;
        }
        bool isEmpty(){
            return !length;
        }
        //打印参数
        void printList(){
            for(int i=1;i<=length;i++){
                cout<<arr[i-1]<<" ";
            }
            cout<<endl;
        }
        //******CURD******
        //插入数据
        bool insertElem(int index,T e){//index：要插入的位置
            if(index<1||index>length+1){
                cout<<"不合法"<<endl;
                return false;
            }
            if(length>=capacity){
                doubleListSize();
            }
            //将index及之后的元素向后移动一位
            for(int i=length;i>=index;i--){
                arr[i]=arr[i-1];
            }
            arr[index-1]=e;
            length++;
           
            return true;
        }
        //读取数据.bool是返回操作结果
        bool getElem(int index,T& e)//&引用表示直接传那个变量本身
        {
            if(index<1||index>length){
                cout<<"illigal"<<endl;
                return false;
            }
            e=arr[index-1];
            return true;
        }
        //查找元素
        int searchElem(T e){
             if(length==0){
                cout<<"empty list";
                return -1;
            }
            for(int i=0;i<length;i++){
                if(arr[i]==e)return i+1;
            }
            cout<<"no information";
            return -1;
        }
        //更新元素
        bool updateElem(int index, T e){
            if(index<=0||index>length){
                cout<<"不合法"<<endl;
                return false;
            }
            arr[index-1]=e;
            return true;
        }
        //删除元素
        bool deleteElem(int index,T& e){
            if(length==0){
                cout<<"空表"<<endl;
                return false;
            }
            if(index<=0||index>length){
                cout<<"不合法"<<endl;
                return false;
            }
            e=arr[index-1];//保存被删除的元素
            for(int i=index-1;i<length-1;i++)
            {
                arr[i]=arr[i+1];
            }
            length--;
            return true;
        }
 
};

int main()
{
```

## 链表
``` cpp
#include<iostream>
using namespace std;
template <typename T>
struct Node{//链表节点
    T data;
    Node<T>* next;
    Node(T val){
        data=val;
        next=nullptr;
    }
    Node(){next=nullptr;}//用于初始化头节点
};

template <typename T>
class Linklist{
    private:
        int length;
        Node<T>* head;
    public:
        Linklist(){
            head=new Node<T>();
            head->next=nullptr;
            length=0;
        }
        Linklist(T* a,int n){
            length=0;
            head=new Node<T>;
            head->next=nullptr;
            for(int i=n-1;i>=0;i--)
            {
                insertElem(1,a[i]);
            }
        }
        ~Linklist(){
            Node<T>* del=head;
            while(del!=nullptr)
            {
                Node<T>* temp=del;
                del=del->next;
                delete temp;
            }
            del=nullptr;
        }
        void printElem(){
            Node<T>* curr=head->next;
            while(curr!=nullptr){
                cout<<curr->data<<" ";
                curr=curr->next;
            }
            cout<<"null"<<endl;
        }
        int getLength(){
            return length;
        }
        bool isEmpty(){
            return length==0;
        }
        //读取某个位置的元素
        bool getElem(int index,T& e){
            if(index<1||index>length){
                cout<<"illigal position"<<endl;
                return false;
            }
            Node<T>* curr=head->next;
            for(int i=1;i<index;i++) curr=curr->next;
            e=curr->data;
            return true;
        }
        //******CURD****** */
        //插入
        bool insertElem(int index, T x){
            if(index<1||index>length+1){
                cout<<"insert wrong";
                return false;
            }
            Node<T>* pre=head;//移动到插入的位置之前一个
            for(int i=1;i<=index-1;i++){
                pre=pre->next;
            }
            Node<T>* newNode=new Node<T>(x);//新建一个节点
            newNode->next=pre->next;
            pre->next=newNode;
            
            length++;
            return true;
        }
        //删除
        bool deleteElem(int index,T& e){//e用于保存删除的数据
            if(index<1||index>length){
                cout<<"delete wrong"<<endl;
                return false;
            }
            Node<T>* pre=head;
            for(int i=1;i<=index-1;i++)pre=pre->next;//到达目标位置前一个节点

            Node<T>* target=pre->next;
            e=target->data;//保留被删除的数据
            pre->next=target->next;//index前一个节点指向index+1
            
            delete target; 
            target=nullptr;
            length--;
            return true;
        }
        //查找：查找是否有这个元素，若有给出其序列号
        int searchElem(T e){
            if(length==0){
                cout<<"empty list";
                return -1;
            }
            Node<T>* curr=head->next;
            for(int i=1;i<=length;i++){
                if(curr->data==e)return i;
                curr=curr->next;
            }
            cout<<"no information";
            return -1;
        }
        //修改：修改某个位置的元素
        bool changeElem(int index,T e){
            if(length==0){
                cout<<"empty list";
                return false;
            }
            if(index<1||index>length){
                cout<<"illigal position";
                return false;
            }
            Node<T>* curr=head;
            for(int i=1;i<=index;i++) curr=curr->next;
            curr->data=e;
            return true;
        }

};
int main()
{
    Linklist<int> list;
    cout<<"test insert "<<endl;
    list.insertElem(1,3);
    list.insertElem(2,2);
    list.insertElem(3,5);
    list.insertElem(4,325);
    list.printElem();
    cout<<"\n test delete"<<endl;
    int data;
    list.deleteElem(3,data);
    list.printElem();
    cout<<"data="<<data;
    cout<<"\n test change"<<endl;
    list.changeElem(2,325);
    list.printElem();
    cout<<"\n test search"<<endl;
    list.searchElem(4);
#ifdef _WIN32
    system("pause");
#endif
    return 0;
}
```
