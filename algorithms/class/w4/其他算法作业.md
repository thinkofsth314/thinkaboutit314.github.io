### 思路1
* **反转字符串，与原串比对（可以暴力可以kmp），匹配不上，翻转串就减少一个字符，继续与原串匹配直到匹配成功。记录匹配成功的字符，再打印原串＋原翻转的，并只打印一遍重复的字符。**
* ``` cpp
  #include <iostream>
  #include <string>
  using namespace std;
  
  void makePalindrome(string s) {
      int n = s.size();
  
      // 建立反转串
      string r;
      for (int i = n - 1; i >= 0; i--) {
          r.push_back(s[i]);
      }
  
      // 从大到小枚举重叠长度
      int len;
      for (len = n; len > 0; len--) {
          bool match = true;
  
          for (int j = 0; j < len; j++) {
              if (s[n - len + j] != r[j]) {
                  match = false;
                  break;
              }
          }
  
          if (match) {
              break;
          }
      }
  
      // 原串全部输出，反转串只输出未重叠部分
      cout << s;
      for (int i = len; i < n; i++) {
          cout << r[i];
      }
      cout << '\n';
  }
  ```
### 思路2
* **只要找出最长后缀就可以满足题设，若s=ABCB，构造s1=BCBA#ABCB#，则找出最后位置的next数组值即可得知最长后缀。找出最长后缀，在s后倒着打印前面的字符即可**
* ``` cpp
  //next 数组
  vector<int> getNext(const string& s) {
      int m = s.length();
      vector<int> next(m, 0);//都初始化为0
      next[0]=-1;
      int i=2;//当前求next值的位置
      int cn=0;//和i-1位置比对的值的下标
      
      while(i<m){
        if(s[i-1]==s[cn])next[i++]= ++cn;//匹配到了，next值就等于匹配到的下标+1
        else if(cn>0)cn=next[cn];//没匹配到，cn往前跳到next[cn]，再匹配
        else next[i++]=0;//cn跳到最前面了，就不能继续跳
      }
  
      return next;//vector可以返回数组本身
      }
  
  string reverse(string s){
      string s0;
      for(int i=s.length()-1;i>=0;i--){
          s0+=s[i];
      }
      return s0;
  }
  
  void makePalindrome(string s){
      int len=s.length();
      string t=reverse(s) +"#"+ s + "#";
      vector<int> next=getNext(t);
      int m=next[2*len+1];
      string s0;
      for(int i=len-1-m;i>=0;i--)s0+=s[i];
      cout<<s<<s0<<endl;
  
  }
  ```

### 暴力
* **枚举长度的因子 → 取对应前缀 → 检查整个字符串是否由它重复构成。**
  - eg：6长的字符，就从头找1，2，3长的字符串，看看是不是等于它们的重复

* ``` cpp
  #include <iostream>
  #include <string>
  using namespace std;
  
  int main() {
      string s;
      cin >> s;
  
      int n = s.size();
  
      for (int k = 1; k <= n; k++) {
          if (n % k != 0) continue;  // 不是因子，跳过
  
          bool match = true;
  
          // 前 k 个字符就是模板，不需要检查自身
          for (int i = k; i < n; i++) {
              if (s[i] != s[i % k]) {
                  match = false;
                  break;
              }
          }
  
          if (match) {
              cout << k << '\n';
              cout << s.substr(0, k);
              break;
          }
      }
  
      return 0;
  }
  ```
